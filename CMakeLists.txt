cmake_minimum_required(VERSION 3.24)
project(chibi_studio LANGUAGES NONE)

# Studio is an authored application. All native code comes from the installed SDK.
find_package(Chibi CONFIG REQUIRED)
option(CHIBI_STUDIO_DISTRIBUTION "Require a pinned native Studio distribution" OFF)
set(CHIBI_STUDIO_AUTHORED ON CACHE INTERNAL "Studio owns authored sources only" FORCE)
set(sdk_manifest "${Chibi_RESOURCE_DIR}/package.json")
if(NOT EXISTS "${sdk_manifest}")
    message(FATAL_ERROR "Studio requires a packaged Chibi SDK")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${sdk_manifest}")
file(READ "${sdk_manifest}" sdk_json)
string(JSON sdk_kind GET "${sdk_json}" kind)
string(JSON sdk_id GET "${sdk_json}" build_id)
if(NOT sdk_kind STREQUAL "sdk" OR NOT sdk_id)
    message(FATAL_ERROR "Studio requires the installed SDK package identity")
endif()
file(SHA256 "${sdk_manifest}" sdk_hash)
set(CHIBI_STUDIO_SDK_BUILD_ID "${sdk_id}" CACHE INTERNAL "SDK package identity" FORCE)
set(CHIBI_STUDIO_SDK_MANIFEST_SHA256 "${sdk_hash}" CACHE INTERNAL "SDK manifest checksum" FORCE)
set(app "${CMAKE_CURRENT_SOURCE_DIR}/projects/studio")
set(CHIBI_STUDIO_ARTIFACT "${CMAKE_BINARY_DIR}/studio/game.chibip" CACHE INTERNAL "Authored application artifact" FORCE)
set(resources "${CHIBI_STUDIO_ARTIFACT}.resources.json")
file(GLOB_RECURSE sources CONFIGURE_DEPENDS "${app}/*")
set(source_inventory "${CMAKE_BINARY_DIR}/studio-inputs.txt")
file(CONFIGURE OUTPUT "${source_inventory}" CONTENT "@sources@\n" @ONLY)

if(CMAKE_SYSTEM_NAME STREQUAL "Emscripten")
    if(CHIBI_STUDIO_DISTRIBUTION)
        message(FATAL_ERROR "Use the web installation for a browser distribution")
    endif()
    set(CHIBI_STUDIO_COLLAB_URL "" CACHE STRING "Trusted HTTP(S) room service origin")
    if(CHIBI_STUDIO_COLLAB_URL AND NOT CHIBI_STUDIO_COLLAB_URL MATCHES "^https?://[^/?#@ ]+/?$")
        message(FATAL_ERROR "Collaboration service must be an HTTP(S) origin")
    endif()
    set(CHIBI_AUTHOR_WEB_DIR "" CACHE PATH "Installed generic web author host and target compiler")
    set(CHIBI_STUDIO_WORKSPACE_ROOT "" CACHE PATH "Optional initial browser workspace; empty opens Home")
    foreach(path chibi-author.js chibi-author.wasm index.html package.json engine.json
        compiler/chibi-compile.js compiler/chibi-compile.wasm compiler/compiler_worker.js
        compiler/chibi-compile-worker.js compiler/chibi-compile-worker.wasm)
        if(NOT EXISTS "${CHIBI_AUTHOR_WEB_DIR}/${path}")
            message(FATAL_ERROR "CHIBI_AUTHOR_WEB_DIR lacks ${path}; package-web-author in the engine first")
        endif()
        set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CHIBI_AUTHOR_WEB_DIR}/${path}")
    endforeach()
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    execute_process(COMMAND "${Python3_EXECUTABLE}" "${Chibi_RESOURCE_DIR}/tools/package_web_author.py"
        --verify --output "${CHIBI_AUTHOR_WEB_DIR}" --sdk-manifest "${sdk_manifest}"
        RESULT_VARIABLE author_verified OUTPUT_VARIABLE author_verification ERROR_VARIABLE author_error)
    if(NOT author_verified EQUAL 0)
        message(FATAL_ERROR "Browser authoring host/SDK verification failed: ${author_error}")
    endif()
    find_program(NODE_EXECUTABLE node REQUIRED)
    set(compiler "${CHIBI_AUTHOR_WEB_DIR}/compiler/chibi-compile.js")
    set(compiler_dependencies "${compiler}" "${CHIBI_AUTHOR_WEB_DIR}/compiler/chibi-compile.wasm")
    set(compile_command "${NODE_EXECUTABLE}" "${compiler}")
else()
    if(CHIBI_STUDIO_DISTRIBUTION AND NOT CMAKE_BUILD_TYPE MATCHES "^(Release|RelWithDebInfo)$")
        message(FATAL_ERROR "Native distribution requires Release or RelWithDebInfo")
    endif()
    if(NOT Chibi_COMPILE_EXECUTABLE OR NOT Chibi_AUTHOR_EXECUTABLE)
        message(FATAL_ERROR "The installed SDK must contain chibi-compile and chibi-author")
    endif()
    set(compiler "${Chibi_COMPILE_EXECUTABLE}")
    set(compiler_dependencies "${compiler}")
    set(compile_command "${compiler}")
endif()

set(app_destination "${CMAKE_BINARY_DIR}/share/chibi/studio")
set(sdk_destination "${CMAKE_BINARY_DIR}/share/chibi/sdk")
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/bin" "${sdk_destination}")
foreach(metadata package.json engine.json)
    configure_file("${Chibi_RESOURCE_DIR}/${metadata}" "${sdk_destination}/${metadata}" COPYONLY)
    install(FILES "${Chibi_RESOURCE_DIR}/${metadata}" DESTINATION share/chibi/sdk)
endforeach()
foreach(resource shaders licenses)
    if(EXISTS "${Chibi_RESOURCE_DIR}/${resource}")
        file(COPY "${Chibi_RESOURCE_DIR}/${resource}" DESTINATION "${sdk_destination}")
        install(DIRECTORY "${Chibi_RESOURCE_DIR}/${resource}/" DESTINATION "share/chibi/sdk/${resource}")
    endif()
endforeach()
add_custom_command(OUTPUT "${CHIBI_STUDIO_ARTIFACT}" "${resources}"
    COMMAND ${compile_command} --project-root "${app}" --output "${CHIBI_STUDIO_ARTIFACT}" --authoring
    COMMAND "${CMAKE_COMMAND}" -E rm -rf "${app_destination}"
    COMMAND "${CMAKE_COMMAND}" -E copy_directory "${app}" "${app_destination}"
    COMMAND "${CMAKE_COMMAND}" -E copy "${CHIBI_STUDIO_ARTIFACT}" "${app_destination}/game.chibip"
    COMMAND "${CMAKE_COMMAND}" -E copy "${resources}" "${app_destination}/game.chibip.resources.json"
    DEPENDS ${sources} ${compiler_dependencies} "${source_inventory}" "${sdk_manifest}"
        "${CMAKE_CURRENT_LIST_FILE}" VERBATIM)
add_custom_target(chibi-studio ALL DEPENDS "${CHIBI_STUDIO_ARTIFACT}" "${resources}")

if(CMAKE_SYSTEM_NAME STREQUAL "Emscripten")
    if(CHIBI_STUDIO_WORKSPACE_ROOT AND NOT EXISTS "${CHIBI_STUDIO_WORKSPACE_ROOT}/project.kdl")
        message(FATAL_ERROR "The initial browser workspace must contain project.kdl")
    endif()
    set(web_destination "${CMAKE_BINARY_DIR}/bin")
    configure_file(cmake/authored_web.cmake.in "${CMAKE_BINARY_DIR}/authored_web.cmake" @ONLY)
    add_custom_command(TARGET chibi-studio POST_BUILD
        COMMAND "${CMAKE_COMMAND}" -P "${CMAKE_BINARY_DIR}/authored_web.cmake" VERBATIM)
    install(DIRECTORY "${web_destination}/" DESTINATION .)
else()
    set(product_suffix "")
    if(WIN32)
        set(product_suffix ".exe")
    endif()
    # Generic game export keeps the SDK's executable-relative engine layout.
    configure_file("${Chibi_RESOURCE_DIR}/engine.json" "${CMAKE_BINARY_DIR}/share/chibi/engine.json" COPYONLY)
    install(FILES "${Chibi_RESOURCE_DIR}/engine.json" DESTINATION share/chibi)
    foreach(resource shaders licenses)
        if(EXISTS "${Chibi_RESOURCE_DIR}/${resource}")
            file(COPY "${Chibi_RESOURCE_DIR}/${resource}" DESTINATION "${CMAKE_BINARY_DIR}/share/chibi")
            install(DIRECTORY "${Chibi_RESOURCE_DIR}/${resource}/" DESTINATION "share/chibi/${resource}")
        endif()
    endforeach()
    get_filename_component(sdk_bin "${Chibi_RESOURCE_DIR}/../../bin" ABSOLUTE)
    foreach(tool chibi chibi-compile chibi-player chibi-author)
        set(binary "${sdk_bin}/${tool}${product_suffix}")
        if(NOT EXISTS "${binary}")
            message(FATAL_ERROR "SDK product is missing: ${binary}")
        endif()
        configure_file("${binary}" "${CMAKE_BINARY_DIR}/bin/${tool}${product_suffix}" COPYONLY)
        install(PROGRAMS "${binary}" DESTINATION bin)
    endforeach()
    if(WIN32)
        set(CHIBI_STUDIO_LAUNCHER "${CMAKE_BINARY_DIR}/bin/chibi-dev.cmd")
        configure_file(cmake/chibi-dev.cmd.in "${CHIBI_STUDIO_LAUNCHER}" @ONLY)
    else()
        set(CHIBI_STUDIO_LAUNCHER "${CMAKE_BINARY_DIR}/bin/chibi-dev")
        configure_file(cmake/chibi-dev.sh.in "${CHIBI_STUDIO_LAUNCHER}" @ONLY)
        file(CHMOD "${CHIBI_STUDIO_LAUNCHER}" PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE GROUP_READ GROUP_EXECUTE WORLD_READ WORLD_EXECUTE)
    endif()
    set(CHIBI_STUDIO_LAUNCHER "${CHIBI_STUDIO_LAUNCHER}" CACHE INTERNAL "Authored application launcher" FORCE)
    install(PROGRAMS "${CHIBI_STUDIO_LAUNCHER}" DESTINATION bin)
    install(DIRECTORY "${app}/" DESTINATION share/chibi/studio)
    install(FILES "${CHIBI_STUDIO_ARTIFACT}" "${resources}" DESTINATION share/chibi/studio)
endif()
